function uploadSingleFile(file) { return new Promise((resolve, reject) => { const isHeic = file.name.toLowerCase().endsWith('.heic') || file.name.toLowerCase().endsWith('.heif') || file.type.includes('heic') || file.type.includes('heif'); // YÖNTEM A: Eğer dosya HEIC ise Canvas dönüştürmesini pas geç, doğrudan oku ve gönder. if (isHeic) { const reader = new FileReader(); reader.onload = function(event) { const base64Data = event.target.result.split(',')[1]; const data = { name: file.name, type: file.type || 'image/heic', base64: base64Data }; sendToGoogleScript(data).then(resolve).catch(reject); }; reader.onerror = (err) => reject(err); reader.readAsDataURL(file); return; } // YÖNTEM B: Diğer standart resimler (JPG, PNG vb.) için mevcut Canvas sıkıştırmanız devam eder. const reader = new FileReader(); reader.onload = function(event) { const img = new Image(); img.onload = function() { const canvas = document.createElement('canvas'); const MAX_WIDTH = 1920; let width = img.width; let height = img.height; if (width > MAX_WIDTH) { height = Math.round((height * MAX_WIDTH) / width); width = MAX_WIDTH; } canvas.width = width; canvas.height = height; const ctx = canvas.getContext('2d'); ctx.drawImage(img, 0, 0, width, height); const dataUrl = canvas.toDataURL('image/jpeg', 0.95); const base64Data = dataUrl.split(',')[1]; const data = { name: file.name, type: 'image/jpeg', base64: base64Data }; sendToGoogleScript(data).then(resolve).catch(reject); }; img.onerror = (err) => reject(err); img.src = event.target.result; }; reader.readAsDataURL(file); }); } // Fetch işlemini tekrar eden kodlardan kaçınmak için ayrı bir yardımcı fonksiyon: function sendToGoogleScript(data) { const scriptURL = 'https://script.google.com/macros/s/AKfycbzgV8bB40dNjzcJznif2Zci_IYxgNSyzAGyLhG76NmkwDGOG8I6B3DXjxBTXS8yxD1J/exec'; return fetch(scriptURL, { method: 'POST', body: JSON.stringify(data), mode: 'no-cors' }); }
